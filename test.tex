Custom background, Python (numerical method):
\begin{codeview}[input, bg=blue!50!black, lang=Python, name=newton.py]
import numpy as np

def newton(f, df, x0, tol=1e-8):
    for i in range(50):
        x1 = x0 - f(x0)/df(x0)
        if abs(x1 - x0) < tol:
            return x1
        x0 = x1
    return x0

print(newton(lambda x: x**2 - 2, lambda x: 2*x, 1.0))
\end{codeview}
\begin{codeview}[output, bg=blue!50!black, prompt=first]
1.414213562373095
converged in 5 iterations
\end{codeview}
